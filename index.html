function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); if (e.parameter.action === "listEmployees") { const sheet = ss.getSheetByName("Employees") || ss.insertSheet("Employees"); const rows = sheet.getDataRange().getValues(); const employees = rows.slice(1).map(r => ({ name: r[0], username: r[1], password: r[2], email: r[3] })); return ContentService.createTextOutput(JSON.stringify({ employees })) .setMimeType(ContentService.MimeType.JSON); } const sheet = ss.getSheetByName("Transactions") || ss.insertSheet("Transactions"); const rows = sheet.getDataRange().getValues(); const transactions = rows.slice(1).map(r => ({ id: r[0], date: r[1], branch: r[2], description: r[3], category: r[4], type: r[5], amount: r[6], enteredBy: r[7] })); return ContentService.createTextOutput(JSON.stringify({ transactions })) .setMimeType(ContentService.MimeType.JSON); } function getAdminSheet(ss) { let sheet = ss.getSheetByName("Admin"); if (!sheet) { sheet = ss.insertSheet("Admin"); sheet.appendRow(["username", "password", "email"]); sheet.appendRow(["admin", "2255", ""]); // default admin — change this from Account Settings after first login } return sheet; } function getResetSheet(ss) { return ss.getSheetByName("ResetCodes") || ss.insertSheet("ResetCodes"); } function doPost(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const body = JSON.parse(e.postData.contents); if (body.action === "login") { const adminSheet = getAdminSheet(ss); const adminRows = adminSheet.getDataRange().getValues(); const adminMatch = adminRows.slice(1).find(r => r[0] === body.username && r[1] === body.password); if (adminMatch) { return ContentService.createTextOutput(JSON.stringify({ ok: true, name: "Admin", role: "admin" })) .setMimeType(ContentService.MimeType.JSON); } const empSheet = ss.getSheetByName("Employees") || ss.insertSheet("Employees"); const empRows = empSheet.getDataRange().getValues(); const empMatch = empRows.slice(1).find(r => r[1] === body.username && r[2] === body.password); if (empMatch) { return ContentService.createTextOutput(JSON.stringify({ ok: true, name: empMatch[0], role: "employee" })) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ ok: false })) .setMimeType(ContentService.MimeType.JSON); } if (body.action === "addEmployee") { const sheet = ss.getSheetByName("Employees") || ss.insertSheet("Employees"); if (sheet.getLastRow() === 0) sheet.appendRow(["name", "username", "password", "email"]); sheet.appendRow([body.employee.name, body.employee.username, body.employee.password, body.employee.email || ""]); return ContentService.createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } if (body.action === "updateAdmin") { const sheet = getAdminSheet(ss); const rows = sheet.getDataRange().getValues(); const rowIndex = rows.findIndex((r, i) => i > 0 && r[0] === body.currentUsername && r[1] === body.currentPassword); if (rowIndex === -1) { return ContentService.createTextOutput(JSON.stringify({ ok: false, error: "Current password incorrect." })) .setMimeType(ContentService.MimeType.JSON); } const row = rowIndex + 1; // 1-indexed for Sheets API if (body.newUsername) sheet.getRange(row, 1).setValue(body.newUsername); if (body.newPassword) sheet.getRange(row, 2).setValue(body.newPassword); if (body.email) sheet.getRange(row, 3).setValue(body.email); return ContentService.createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } if (body.action === "forgotPassword") { const adminSheet = getAdminSheet(ss); const adminRows = adminSheet.getDataRange().getValues(); const adminRow = adminRows.slice(1).find(r => r[0] === body.username); const empSheet = ss.getSheetByName("Employees") || ss.insertSheet("Employees"); const empRows = empSheet.getDataRange().getValues(); const empRow = empRows.slice(1).find(r => r[1] === body.username); const email = adminRow ? adminRow[2] : (empRow ? empRow[3] : null); if (!email) { return ContentService.createTextOutput(JSON.stringify({ ok: false, error: "No email on file for this account." })) .setMimeType(ContentService.MimeType.JSON); } const code = Math.floor(100000 + Math.random() * 900000).toString(); const resetSheet = getResetSheet(ss); if (resetSheet.getLastRow() === 0) resetSheet.appendRow(["username", "code", "expires"]); resetSheet.appendRow([body.username, code, new Date(Date.now() + 15 * 60 * 1000).toISOString()]); MailApp.sendEmail(email, "Zion Fresh — Password Reset Code", "Your password reset code is: " + code + "\n\nThis code expires in 15 minutes. If you didn't request this, you can ignore this email."); return ContentService.createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } if (body.action === "resetPassword") { const resetSheet = getResetSheet(ss); const resetRows = resetSheet.getDataRange().getValues(); const rowIndex = resetRows.findIndex((r, i) => i > 0 && r[0] === body.username && r[1] === body.code && new Date(r[2]) > new Date()); if (rowIndex === -1) { return ContentService.createTextOutput(JSON.stringify({ ok: false, error: "Incorrect or expired code." })) .setMimeType(ContentService.MimeType.JSON); } const adminSheet = getAdminSheet(ss); const adminRows = adminSheet.getDataRange().getValues(); const adminRowIdx = adminRows.findIndex((r, i) => i > 0 && r[0] === body.username); if (adminRowIdx > -1) { adminSheet.getRange(adminRowIdx + 1, 2).setValue(body.newPassword); } else { const empSheet = ss.getSheetByName("Employees") || ss.insertSheet("Employees"); const empRows = empSheet.getDataRange().getValues(); const empRowIdx = empRows.findIndex((r, i) => i > 0 && r[1] === body.username); if (empRowIdx > -1) empSheet.getRange(empRowIdx + 1, 3).setValue(body.newPassword); } resetSheet.deleteRow(rowIndex + 1); return ContentService.createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } const sheet = ss.getSheetByName("Transactions") || ss.insertSheet("Transactions"); if (sheet.getLastRow() === 0) { sheet.appendRow(["id", "date", "branch", "description", "category", "type", "amount", "enteredBy"]); } const entry = body.entry; sheet.appendRow([Date.now(), entry.date, entry.branch, entry.description, entry.category, entry.type, entry.amount, entry.enteredBy]); return ContentService.createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); }
