function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); if (e.parameter.action === "listEmployees") { const sheet = ss.getSheetByName("Employees") || ss.insertSheet("Employees"); const rows = sheet.getDataRange().getValues(); const employees = rows.slice(1).map(r => ({ name: r[0], username: r[1], password: r[2] })); return ContentService.createTextOutput(JSON.stringify({ employees })) .setMimeType(ContentService.MimeType.JSON); } const sheet = ss.getSheetByName("Transactions") || ss.insertSheet("Transactions"); const rows = sheet.getDataRange().getValues(); const transactions = rows.slice(1).map(r => ({ id: r[0], date: r[1], branch: r[2], description: r[3], category: r[4], type: r[5], amount: r[6], enteredBy: r[7] })); return ContentService.createTextOutput(JSON.stringify({ transactions })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const body = JSON.parse(e.postData.contents); if (body.action === "login") { const sheet = ss.getSheetByName("Employees") || ss.insertSheet("Employees"); const rows = sheet.getDataRange().getValues(); const match = rows.slice(1).find(r => r[1] === body.username && r[2] === body.password); if (match) { return ContentService.createTextOutput(JSON.stringify({ ok: true, name: match[0] })) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ ok: false })) .setMimeType(ContentService.MimeType.JSON); } if (body.action === "addEmployee") { const sheet = ss.getSheetByName("Employees") || ss.insertSheet("Employees"); if (sheet.getLastRow() === 0) sheet.appendRow(["name", "username", "password"]); sheet.appendRow([body.employee.name, body.employee.username, body.employee.password]); return ContentService.createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } const sheet = ss.getSheetByName("Transactions") || ss.insertSheet("Transactions"); if (sheet.getLastRow() === 0) { sheet.appendRow(["id", "date", "branch", "description", "category", "type", "amount", "enteredBy"]); } const entry = body.entry; sheet.appendRow([Date.now(), entry.date, entry.branch, entry.description, entry.category, entry.type, entry.amount, entry.enteredBy]); return ContentService.createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); }
